cmake_minimum_required(VERSION 3.20)
project(nadoc_vr_capture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
include(CTest)
add_library(nadoc_present_capture SHARED present_capture.cpp)
target_include_directories(nadoc_present_capture PRIVATE ${Vulkan_INCLUDE_DIRS})
target_link_libraries(nadoc_present_capture PRIVATE Threads::Threads ZLIB::ZLIB ${CMAKE_DL_LIBS})
target_compile_options(nadoc_present_capture PRIVATE -Wall -Wextra -Wno-missing-field-initializers -fvisibility=hidden)
target_link_options(nadoc_present_capture PRIVATE -Wl,-Bsymbolic)
configure_file(VkLayer_NADOC_present_capture.json.in VkLayer_NADOC_present_capture.json @ONLY)
find_package(X11)
if(X11_FOUND)
  add_executable(nadoc_capture_probe probe.cpp)
  target_link_libraries(nadoc_capture_probe PRIVATE Vulkan::Vulkan X11::X11)
endif()
if(BUILD_TESTING)
  add_executable(nadoc_capture_contract_test contract_test.cpp)
  target_include_directories(nadoc_capture_contract_test PRIVATE ${Vulkan_INCLUDE_DIRS})
  target_link_libraries(nadoc_capture_contract_test PRIVATE Threads::Threads ZLIB::ZLIB ${CMAKE_DL_LIBS})
  target_compile_options(nadoc_capture_contract_test PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
  add_test(NAME nadoc_capture_contract COMMAND nadoc_capture_contract_test ${CMAKE_CURRENT_BINARY_DIR}/contract-artifacts)
endif()
